function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["ID", "الاسم", "الصف والشعبة", "وقت الدخول", "المدة", "الشرائح", "المجاب", "الصحيح", "الجهاز"]);
  }
  var d = JSON.parse(e.postData.contents);
  sheet.appendRow([d.id, d.name, d.classSection, d.loginTime, d.durationSec, d.slidesVisited, d.answeredCount, d.correctCount, d.device]);
  return ContentService.createTextOutput(JSON.stringify({status:"ok"})).setMimeType(ContentService.MimeType.JSON);
}
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  return ContentService.createTextOutput(JSON.stringify({rows: rows})).setMimeType(ContentService.MimeType.JSON);
}
